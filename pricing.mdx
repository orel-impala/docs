export const PriceCalculator = () => {
  const W = ["asap", "standard", "flex", "batch"];
  const M = [
    { n: "GLM-5.3", p: [[1.1, 0.2, 3.5], [0.55, 0.1, 1.75], [0.385, 0.07, 1.225], [0.33, 0.06, 1.05]] },
    { n: "GLM-5.2", p: [[0.95, 0.11, 3.4], [0.475, 0.055, 1.7], [0.3325, 0.0385, 1.19], [0.285, 0.033, 1.02]] },
    { n: "DeepSeek V4.1 Flash", p: [[0.25, 0.025, 1.0], [0.125, 0.0125, 0.5], [0.0875, 0.0088, 0.35], [0.075, 0.0028, 0.285]] },
    { n: "Gemma 4 31B", p: [[0.14, 0.06, 0.4], [0.07, 0.03, 0.2], [0.049, 0.021, 0.14], [0.042, 0.018, 0.12]] },
    { n: "gpt-oss-120b", p: [[0.06, 0.03, 0.4], [0.03, 0.015, 0.2], [0.021, 0.0105, 0.14], [0.018, 0.0071, 0.12]] },
    { n: "Kimi K3", p: [[2.25, 0.225, 11.25], [1.125, 0.1125, 5.625], [0.7875, 0.0788, 3.9375], [0.675, 0.0675, 3.375]] },
  ];
  const [mi, setMi] = useState(0);
  const [tok, setTok] = useState(2);
  const [t, setT] = useState(1);
  const m = M[mi];
  const px = W.map((_, w) => m.p[w][tok]);
  const fp = (v) => {
    let s = v.toFixed(4).replace(/0+$/, "");
    if ((s.split(".")[1] || "").length < 2) s = v.toFixed(2);
    return "$" + s;
  };
  const dd = [0, 1, 2].map((k) => px[k + 1] - px[k]);
  const sl = [0, 1, 2, 3].map((k) => (k === 0 || k === 3 ? 0 : dd[k - 1] * dd[k] <= 0 ? 0 : 2 / (1 / dd[k - 1] + 1 / dd[k])));
  sl[0] = Math.min(0, 1.5 * dd[0] - 0.5 * sl[1]); sl[3] = Math.min(0, 1.5 * dd[2] - 0.5 * sl[2]);
  const f = (x) => { const k = Math.min(2, Math.floor(x)), u = x - k, u2 = u * u, u3 = u2 * u; return (2 * u3 - 3 * u2 + 1) * px[k] + (u3 - 2 * u2 + u) * sl[k] + (-2 * u3 + 3 * u2) * px[k + 1] + (u3 - u2) * sl[k + 1]; };
  const L = 56, R = 610, TOP = 36, BASE = 176;
  const nice = (v) => { const e = Math.pow(10, Math.floor(Math.log10(v))); const n = v / e; return (n <= 1 ? 1 : n <= 2 ? 2 : n <= 2.5 ? 2.5 : n <= 5 ? 5 : 10) * e; };
  const max = nice(Math.max(...px) || 1);
  const ticks = [0, 0.25, 0.5, 0.75, 1].map((r) => r * max);
  const USE = ["interactive", "agents, pipelines", "evals, enrichment", "offline jobs"];
  const X = (x) => L + (x / 3) * (R - L);
  const Y = (v) => BASE - (v / max) * (BASE - TOP);
  const line = Array.from({ length: 121 }, (_, i) => (i ? "L" : "M") + X(i / 40).toFixed(1) + " " + Y(f(i / 40)).toFixed(1)).join(" ");
  const area = line + " L" + R + " " + BASE + " L" + L + " " + BASE + " Z";
  const snapT = (x) => { const r = Math.round(x); return Math.abs(x - r) < 0.12 ? r : x; };
  const move = (e) => {
    const b = e.currentTarget.getBoundingClientRect();
    const vx = ((e.clientX - b.left) / b.width) * 640;
    setT(snapT(Math.max(0, Math.min(3, ((vx - L) / (R - L)) * 3))));
  };
  const key = (e) => {
    if (e.key === "ArrowRight") { e.preventDefault(); setT((v) => snapT(Math.min(3, v + 0.25))); }
    if (e.key === "ArrowLeft") { e.preventDefault(); setT((v) => snapT(Math.max(0, v - 0.25))); }
  };
  const onTier = Number.isInteger(t);
  const lo = Math.floor(t), hi = Math.min(3, Math.ceil(t));
  const cx = X(t), cy = Y(onTier ? px[t] : f(t));
  const label = onTier ? W[t] + " · " + fp(px[t]) : "custom · " + fp(px[hi]) + "–" + fp(px[lo]);
  const lw = label.length * 6.6 + 20;
  const lx = Math.max(L - 10, Math.min(R + 10 - lw, cx - lw / 2));
  const muted = "rgb(var(--gray-500))";
  const hair = "rgb(var(--gray-500) / 0.25)";
  const accent = "rgb(var(--primary))";
  const seg = (on) => ({ padding: "3px 10px", borderRadius: 6, border: "none", background: on ? "rgb(var(--gray-500) / 0.18)" : "transparent", color: on ? "inherit" : muted, fontSize: 13, cursor: "pointer" });
  return (
    <div className="not-prose" style={{ border: "1px solid " + hair, borderRadius: 14, padding: "18px 20px 10px", margin: "20px 0" }}>
      <div style={{ display: "flex", flexWrap: "wrap", gap: 12, alignItems: "center", justifyContent: "space-between" }}>
        <select value={mi} onChange={(e) => setMi(Number(e.target.value))} aria-label="Model" style={{ padding: "4px 8px", borderRadius: 6, border: "1px solid " + hair, background: "transparent", color: "inherit", fontSize: 13 }}>
          {M.map((x, i) => <option key={x.n} value={i}>{x.n}</option>)}
        </select>
        <div style={{ display: "flex", gap: 2, padding: 2, borderRadius: 8, border: "1px solid " + hair }}>
          {[0, 2].map((k) => <button key={k} type="button" onClick={() => setTok(k)} style={seg(tok === k)}>{k === 0 ? "Input" : "Output"}</button>)}
        </div>
      </div>
      <svg viewBox="0 0 640 228" onPointerMove={move} onPointerDown={move} onKeyDown={key} tabIndex={0} role="img" aria-label={"Price per 1M " + (tok ? "output" : "input") + " tokens by window: " + label} style={{ width: "100%", display: "block", marginTop: 8, cursor: "crosshair", touchAction: "pan-y", outline: "none" }}>
        <defs>
          <linearGradient id="pcfill" x1="0" y1="0" x2="0" y2="1">
            <stop offset="0" stopColor="rgb(var(--primary))" stopOpacity="0.22" />
            <stop offset="1" stopColor="rgb(var(--primary))" stopOpacity="0" />
          </linearGradient>
        </defs>
        {ticks.map((v, i) => (
          <g key={i}>
            <line x1={L} y1={Y(v)} x2={R} y2={Y(v)} stroke={hair} strokeDasharray={i ? "2 4" : "0"} />
            <text x={L - 8} y={Y(v) + 4} textAnchor="end" fontSize="11" fill={muted}>{"$" + (+v.toPrecision(3))}</text>
          </g>
        ))}
        <path d={area} fill="url(#pcfill)" />
        <path d={line} fill="none" stroke={accent} strokeWidth="2" strokeLinecap="round" />
        {px.map((v, w) => (
          <g key={w}>
            <circle cx={X(w)} cy={Y(v)} r="3" fill={accent} fillOpacity="0.55" />
            <text x={X(w)} y={BASE + 20} textAnchor={w === 0 ? "start" : w === 3 ? "end" : "middle"} fontSize="12" fill={t === w ? "currentColor" : muted} fontFamily="var(--font-mono, monospace)">{W[w]}</text>
            <text x={X(w)} y={BASE + 36} textAnchor={w === 0 ? "start" : w === 3 ? "end" : "middle"} fontSize="11" fill={muted}>{USE[w]}</text>
          </g>
        ))}
        <line x1={cx} y1={cy} x2={cx} y2={BASE} stroke={accent} strokeOpacity="0.35" strokeDasharray="2 3" />
        <circle cx={cx} cy={cy} r="5" fill={accent} />
        <g transform={"translate(" + lx + "," + Math.max(2, cy - 34) + ")"}>
          <rect width={lw} height="22" rx="6" fill="rgb(var(--gray-500) / 0.14)" />
          <text x={lw / 2} y="15" textAnchor="middle" fontSize="12" fill="currentColor" fontWeight={onTier ? 600 : 400}>{label}</text>
        </g>
      </svg>
      <div style={{ color: muted, fontSize: 12, textAlign: "right" }}>USD per 1M {tok ? "output" : "input"} tokens</div>
    </div>
  );
};

Prices are in USD per million tokens. Each request is billed at the rate of its [completion window](/completion-windows): the longer the window, the lower the price.
