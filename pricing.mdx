export const PriceCalculator = () => {
  const W = ["asap", "standard", "flex", "batch"];
  const M = [
    { n: "GLM-5.3", p: [[1.1, 0.2, 3.5], [0.55, 0.1, 1.75], [0.385, 0.07, 1.225], [0.33, 0.06, 1.05]] },
    { n: "GLM-5.2", p: [[0.95, 0.11, 3.4], [0.475, 0.055, 1.7], [0.3325, 0.0385, 1.19], [0.285, 0.033, 1.02]] },
    { n: "DeepSeek V4.1 Flash", p: [[0.25, 0.025, 1.0], [0.125, 0.0125, 0.5], [0.0875, 0.0088, 0.35], [0.075, 0.0028, 0.285]] },
    { n: "Gemma 4 31B", p: [[0.14, 0.06, 0.4], [0.07, 0.03, 0.2], [0.049, 0.021, 0.14], [0.042, 0.018, 0.12]] },
    { n: "gpt-oss-120b", p: [[0.06, 0.03, 0.4], [0.03, 0.015, 0.2], [0.021, 0.0105, 0.14], [0.018, 0.0071, 0.12]] },
    { n: "Kimi K3", p: [[2.25, 0.225, 11.25], [1.125, 0.1125, 5.625], [0.7875, 0.0788, 3.9375], [0.675, 0.0675, 3.375]] },
  ];
  const TALK = "https://meetings-eu1.hubspot.com/ran-friedman";
  const [mi, setMi] = useState(0);
  const [tok, setTok] = useState(2);
  const [pos, setPos] = useState(100);
  const m = M[mi];
  const px = W.map((_, w) => m.p[w][tok]);
  const fp = (v) => {
    let s = v.toFixed(4).replace(/0+$/, "");
    if ((s.split(".")[1] || "").length < 2) s = v.toFixed(2);
    return "$" + s;
  };
  const snap = (v) => { const t = Math.round(v / 100); return Math.abs(v - t * 100) <= 6 ? t * 100 : v; };
  const at = pos / 100;
  const lo = Math.floor(at), hi = Math.min(3, Math.ceil(at));
  const onTier = lo === hi;
  const max = Math.max(...px) || 1;
  const X = (t) => 50 + t * 150;
  const Y = (v) => 150 - (v / max) * 115;
  const dd = [0, 1, 2].map((k) => px[k + 1] - px[k]);
  const sl = [0, 1, 2, 3].map((k) => (k === 0 || k === 3 ? 0 : dd[k - 1] * dd[k] <= 0 ? 0 : 2 / (1 / dd[k - 1] + 1 / dd[k])));
  sl[0] = Math.min(0, 1.5 * dd[0] - 0.5 * sl[1]); sl[3] = Math.min(0, 1.5 * dd[2] - 0.5 * sl[2]);
  const f = (t) => { const k = Math.min(2, Math.floor(t)), u = t - k, u2 = u * u, u3 = u2 * u; return (2 * u3 - 3 * u2 + 1) * px[k] + (u3 - 2 * u2 + u) * sl[k] + (-2 * u3 + 3 * u2) * px[k + 1] + (u3 - u2) * sl[k + 1]; };
  const yAt = f(at);
  const line = Array.from({ length: 91 }, (_, i) => (i ? "L" : "M") + X(i / 30).toFixed(1) + " " + Y(f(i / 30)).toFixed(1)).join(" ");
  const area = line + " L" + X(3) + " 150 L" + X(0) + " 150 Z";
  const muted = "rgb(var(--gray-500))";
  const border = "1px solid rgb(var(--gray-500) / 0.3)";
  const accent = "rgb(var(--primary))";
  const pill = (on) => ({ padding: "4px 14px", borderRadius: 999, border: on ? "1px solid " + accent : border, background: on ? "rgb(var(--primary) / 0.12)" : "transparent", color: "inherit", fontSize: 14, cursor: "pointer" });
  const TOK = ["input", "cached input", "output"];
  return (
    <div className="not-prose" style={{ border: border, borderRadius: 16, padding: 24, margin: "16px 0" }}>
      <div style={{ fontSize: 20, fontWeight: 600 }}>Price per window</div>
      <div style={{ color: muted, fontSize: 14, marginTop: 4 }}>Drag along the curve. The four windows have list prices; anything in between is a custom window we can set up for you.</div>
      <div style={{ display: "flex", flexWrap: "wrap", gap: 8, marginTop: 16, alignItems: "center" }}>
        <select value={mi} onChange={(e) => setMi(Number(e.target.value))} aria-label="Model" style={{ padding: "6px 10px", borderRadius: 8, border: border, background: "transparent", color: "inherit", fontSize: 14 }}>
          {M.map((x, i) => <option key={x.n} value={i}>{x.n}</option>)}
        </select>
        {[0, 2].map((t) => <button key={t} type="button" onClick={() => setTok(t)} style={pill(tok === t)}>{TOK[t]}</button>)}
      </div>
      <svg viewBox="0 0 560 185" style={{ width: "100%", maxWidth: 560, marginTop: 16, display: "block" }} role="img" aria-label="Price per million tokens by completion window">
        <path d={area} fill="rgb(var(--primary) / 0.10)" />
        <path d={line} fill="none" stroke={accent} strokeWidth="2" strokeDasharray="5 4" />
        <line x1={X(0)} y1="150" x2={X(3)} y2="150" stroke="rgb(var(--gray-500) / 0.4)" />
        {px.map((v, w) => (
          <g key={w} onClick={() => setPos(w * 100)} style={{ cursor: "pointer" }}>
            <circle cx={X(w)} cy={Y(v)} r="5" fill={accent} />
            <text x={X(w)} y={Y(v) - 10} textAnchor="middle" fontSize="12" fill="currentColor">{fp(v)}</text>
            <text x={X(w)} y="170" textAnchor="middle" fontSize="12" fill="rgb(var(--gray-500))" fontFamily="monospace">{W[w]}</text>
          </g>
        ))}
        {!onTier && <line x1={X(at)} y1={Y(yAt)} x2={X(at)} y2="150" stroke={accent} strokeOpacity="0.5" />}
        {!onTier && <circle cx={X(at)} cy={Y(yAt)} r="7" fill="none" stroke={accent} strokeWidth="2" />}
        {onTier && <circle cx={X(lo)} cy={Y(px[lo])} r="9" fill="none" stroke={accent} strokeWidth="2" />}
      </svg>
      <input type="range" min="0" max="300" step="1" value={pos} onChange={(e) => setPos(snap(Number(e.target.value)))} aria-label="Completion window" style={{ width: "calc(100% - 20px)", maxWidth: 470, margin: "0 0 0 45px", accentColor: accent }} />
      <div style={{ marginTop: 16, fontSize: 15, minHeight: 48 }}>
        {onTier ? (
          <span><code>{W[lo]}</code> · <span style={{ fontSize: 22, fontWeight: 700 }}>{fp(px[lo])}</span> <span style={{ color: muted }}>per 1M {TOK[tok]} tokens, list price</span></span>
        ) : (
          <span>Custom window between <code>{W[lo]}</code> and <code>{W[hi]}</code> · <span style={{ fontSize: 22, fontWeight: 700 }}>{fp(px[hi])}–{fp(px[lo])}</span> <span style={{ color: muted }}>per 1M {TOK[tok]} tokens</span>
            <br /><a href={TALK} target="_blank" style={{ color: accent, fontWeight: 600 }}>Talk to us to set a custom window →</a></span>
        )}
      </div>
    </div>
  );
};

Prices are in USD per million tokens. Each request is billed at the rate of its [completion window](/completion-windows): the longer the window, the lower the price.
